cmake_minimum_required(VERSION 3.22)
project(CellBook VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CELLBOOK_CORE_ONLY "Build formula tests without Qt" OFF)
add_library(cellbook_core src/core.cpp)
target_include_directories(cellbook_core PUBLIC src)
enable_testing()
add_executable(core_test tests/core_test.cpp)
target_link_libraries(core_test PRIVATE cellbook_core)
add_test(NAME formulas COMMAND core_test)
if(CELLBOOK_CORE_ONLY)
  return()
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets PrintSupport Gui)
set(QT_VERSION_MAJOR 6)
set(CMAKE_AUTOMOC ON)
include(FetchContent)
FetchContent_Declare(qxlsx
  GIT_REPOSITORY https://github.com/QtExcel/QXlsx.git
  GIT_TAG 3192b808a283f8eb18c97b9a087bf085054e9bb0
  SOURCE_SUBDIR QXlsx)
FetchContent_MakeAvailable(qxlsx)
# QXlsx 1.4.9 uses startsWith("/") in its reader; allow its ASCII string conversion.
get_target_property(qxlsx_definitions QXlsx COMPILE_DEFINITIONS)
list(REMOVE_ITEM qxlsx_definitions QT_NO_CAST_FROM_ASCII)
set_property(TARGET QXlsx PROPERTY COMPILE_DEFINITIONS "${qxlsx_definitions}")
add_executable(CellBook WIN32 src/main.cpp src/ui.cpp src/io.cpp)
target_link_libraries(CellBook PRIVATE cellbook_core Qt6::Widgets Qt6::PrintSupport QXlsx::QXlsx)
if(MSVC)
  target_compile_options(CellBook PRIVATE /utf-8)
  target_compile_options(cellbook_core PRIVATE /utf-8)
  target_compile_options(core_test PRIVATE /utf-8)
endif()
add_executable(io_test tests/io_test.cpp src/io.cpp)
target_link_libraries(io_test PRIVATE cellbook_core Qt6::Widgets QXlsx::QXlsx)
if(MSVC)
  target_compile_options(io_test PRIVATE /utf-8)
endif()
add_test(NAME file_roundtrip COMMAND io_test)
add_test(NAME gui_smoke COMMAND CellBook --self-test)
set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
